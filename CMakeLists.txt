cmake_minimum_required(VERSION 3.20)
project(openwrt-njupt-login VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(CURL 7.66 REQUIRED)

add_library(njupt_core src/config.cpp src/protocol.cpp src/controller.cpp)
target_include_directories(njupt_core PUBLIC src)
add_library(njupt_http src/http.cpp)
target_link_libraries(njupt_http PUBLIC njupt_core CURL::libcurl)
add_executable(njupt-login src/main.cpp)
target_link_libraries(njupt-login PRIVATE njupt_http)
if(WIN32)
  target_compile_definitions(njupt_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(njupt_core PUBLIC ws2_32)
endif()
foreach(target njupt_core njupt_http njupt-login)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()

include(CTest)
if(BUILD_TESTING)
  add_executable(core-tests tests/core_tests.cpp)
  target_link_libraries(core-tests PRIVATE njupt_core)
  if(MSVC)
    target_compile_options(core-tests PRIVATE /W4 /WX /utf-8 /permissive-)
  else()
    target_compile_options(core-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME core COMMAND core-tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME protocol-integration COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:njupt-login>)
    add_test(NAME network-scripts COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/network_scripts.py)
  endif()
endif()
install(TARGETS njupt-login RUNTIME DESTINATION bin)
